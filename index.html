async function handleTrack() { const waybill = waybillInput.value.trim(); const phone = trackPhoneInput.value.trim(); if (!waybill) { showToast('请输入运单号'); return; } if (waybill.length < 8 || waybill.length > 30) { showToast('运单号长度应为 8~30 位'); return; } if (phone && !/^1[3-9]\d{9}$/.test(phone)) { showToast('关联手机号格式不正确'); return; } trackBtn.disabled = true; trackBtn.innerHTML = '反查中'; try { const res = await fetch('/api/trackquery', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ waybill, phone: phone || undefined }), }); const json = await res.json(); if (!res.ok) { showToast(json.error || '反查失败'); return; } if (!json.ok) { showToast('未查到取件码：' + (json.error || '快递鸟未返回取件码')); return; } const d = json.data; if (d.pickupCode) showToast(`取件码：${d.pickupCode}` + (json.mock ? '（mock）' : '')); else showToast('快递已派送，但轨迹中暂无取件码'); if (phone) { phoneInput.value = phone; await handleQuery(); } else { const item = { company: d.company, waybill: d.waybill, pickupCode: d.pickupCode, station: d.station, traces: d.traces, trackState: d.state, status: d.pickupCode ? 'arrived' : d.state === '3' ? 'signed' : d.state === '2' ? 'in_transit' : d.state === '4' ? 'problem' : 'pending', receivedAt: new Date().toISOString(), source: 'track', }; state.allItems = [item]; state.filter = 'all'; document.querySelectorAll('.filter-tab').forEach(btn => { btn.classList.toggle('active', btn.dataset.filter === 'all'); }); renderList(); } } catch (e) { showToast('网络错误：' + e.message); } finally { trackBtn.disabled = false; trackBtn.textContent = '反查取件码'; } }
